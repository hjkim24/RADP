% Sections are included from sections/*.tex so multiple authors can work
% in parallel without main.tex merge conflicts. The section bodies use only
% standard LaTeX (\section, \cite, \ref, figure/table/algorithm), so they are
% class-agnostic and needed no changes in the acmart -> IEEEtran port.

\documentclass[journal]{IEEEtran}

% --- Packages ----------------------------------------------------------
% IEEEtran (unlike acmart) auto-loads almost nothing, so every package the
% body relies on is listed explicitly here.
\usepackage{cite}                 % IEEE-style citation compression [1]-[3]
\usepackage[T1]{fontenc}          % 8-bit T1 output encoding, required by newtx
\usepackage{amsmath,amssymb,amsfonts}
% Times-family fonts: IEEEtran defaults to Computer Modern (light/"plain"
% looking); published IEEE Transactions use Times. newtx gives Times text +
% matching Times math and MUST load after amsmath/amssymb.
\usepackage{newtxtext}            % Times text
\usepackage{newtxmath}            % matching Times math
\usepackage{algorithm}
\usepackage{algpseudocode}
\usepackage{graphicx}
\usepackage{booktabs}
\usepackage{multirow}
% NOTE: do not load caption/subcaption. They override IEEEtran's native caption
% format ("Fig. 1." and small-caps "TABLE I" headers become "Fig. 1:" /
% "TABLE I:"). No sub-figures are used, so nothing is lost by omitting them.
\usepackage{listings}
\usepackage{xcolor}               % for \todo color
\usepackage{xspace}
\usepackage{textcomp}
% hyperref last; hidelinks keeps clickable refs without coloured boxes,
% which is the convention for IEEE camera-ready.
\usepackage[hidelinks]{hyperref}

\graphicspath{{figures/}}

% --- Typesetting tweaks ------------------------------------------------
% emergencystretch lets TeX widen interword spacing as a last resort to
% absorb the sub-20pt overfull boxes that \texttt{} identifiers create in
% two-column layout, without the whole-paper spread that global \sloppy
% causes. Float separations are trimmed so a wide table + algorithm don't
% cost an extra page.
\setlength{\emergencystretch}{2em}
\setlength{\textfloatsep}{8pt plus 2pt minus 2pt}
\setlength{\dbltextfloatsep}{8pt plus 2pt minus 2pt}
\setlength{\intextsep}{6pt plus 2pt minus 2pt}
